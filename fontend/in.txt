      // ============================================================
      // KHI MAP LOAD XONG → THÊM TẤT CẢ LAYER
      // ============================================================
      map.on("load", () => {
        const status = document.getElementById("status");
        status.textContent = `Đang thêm ${LAYERS.length} layer...`;

        LAYERS.forEach((cfg) => {
          const sourceId = cfg.id + "-src";
          const layerId = cfg.id + "-lyr";
          const visible = cfg.default !== false;

          // 1. Source
          map.addSource(sourceId, {
            type: "vector",
            tiles: [`${TILE_URL}/public.${cfg.id}/{z}/{x}/{y}.pbf`],
            minzoom: 0,
            maxzoom: 19,
          });

          // 2. Layer chính
          const layerDef = {
            id: layerId,
            type: cfg.type,
            source: sourceId,
            "source-layer": `public.${cfg.id}`,
            layout: { visibility: visible ? "visible" : "none" },
            paint: cfg.paint || {},
          };
          if (cfg.minzoom) layerDef.minzoom = cfg.minzoom;
          map.addLayer(layerDef);

          // 3. Nếu có outline (ranh giới hành chính) → thêm line layer chồng lên
          if (cfg.outlinePaint) {
            map.addLayer({
              id: layerId + "-outline",
              type: "line",
              source: sourceId,
              "source-layer": `public.${cfg.id}`,
              layout: { visibility: visible ? "visible" : "none" },
              paint: cfg.outlinePaint,
            });
          }

          // 4. Sự kiện click để hiện popup
          map.on("click", layerId, (e) => {
            const p = e.features[0].properties;
            showPopup(e.lngLat, p, cfg.label);
          });
          map.on(
            "mouseenter",
            layerId,
            () => (map.getCanvas().style.cursor = "pointer"),
          );
          map.on(
            "mouseleave",
            layerId,
            () => (map.getCanvas().style.cursor = ""),
          );
        });

        // 5. Xây dựng panel checkbox
        buildLayerPanel();
        status.textContent = `✅ Đã tải ${LAYERS.length} layer`;
        setTimeout(() => (status.style.display = "none"), 3000);
      });

      // ============================================================
      // POPUP
      // ============================================================
      function showPopup(lngLat, props, layerLabel) {
        const parts = [];
        if (props.name)
          parts.push(`<div class="popup-name">${esc(props.name)}</div>`);
        if (props.fclass)
          parts.push(
            `<div class="popup-prop"><b>Loại:</b> ${esc(props.fclass)}</div>`,
          );
        if (props.type && props.type !== props.fclass) {
          parts.push(
            `<div class="popup-prop"><b>Phân loại:</b> ${esc(props.type)}</div>`,
          );
        }
        parts.push(`<div class="popup-layer">Lớp: ${esc(layerLabel)}</div>`);

        new maplibregl.Popup({ closeButton: true, maxWidth: "300px" })
          .setLngLat(lngLat)
          .setHTML(`<div class="map-popup">${parts.join("")}</div>`)
          .addTo(map);
      }

      function esc(s) {
        return String(s).replace(
          /[&<>"']/g,
          (c) =>
            ({
              "&": "&amp;",
              "<": "&lt;",
              ">": "&gt;",
              '"': "&quot;",
              "'": "&#39;",
            })[c],
        );
      }

      // ============================================================
      // PANEL BẬT/TẮT LAYER
      // ============================================================
      function buildLayerPanel() {
        const list = document.getElementById("layer-list");
        const groupLabels = {
          Polygon: "🔷 Vùng (Polygon)",
          Line: "📏 Đường (Line)",
          Point: "📍 Điểm (Point)",
        };
        let currentGroup = null;

        LAYERS.forEach((cfg) => {
          if (cfg.group !== currentGroup) {
            currentGroup = cfg.group;
            const title = document.createElement("div");
            title.className = "layer-group-title";
            title.textContent = groupLabels[cfg.group] || cfg.group;
            list.appendChild(title);
          }

          const row = document.createElement("label");
          row.className = "layer-item";
          const cb = document.createElement("input");
          cb.type = "checkbox";
          cb.checked = cfg.default !== false;
          cb.onchange = () => toggleLayer(cfg.id, cb.checked);
          row.appendChild(cb);
          row.appendChild(document.createTextNode(" " + cfg.label));
          list.appendChild(row);
        });
      }

      function toggleLayer(id, visible) {
        const ids = [id + "-lyr", id + "-lyr-outline"];
        ids.forEach((lid) => {
          if (map.getLayer(lid)) {
            map.setLayoutProperty(
              lid,
              "visibility",
              visible ? "visible" : "none",
            );
          }
        });
      }

      function toggleAll(visible) {
        LAYERS.forEach((cfg) => toggleLayer(cfg.id, visible));
        document
          .querySelectorAll("#layer-list input[type=checkbox]")
          .forEach((cb) => (cb.checked = visible));
      }
